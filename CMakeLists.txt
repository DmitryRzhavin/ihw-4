cmake_minimum_required(VERSION 3.16)
project(restaurant_network C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(COMMON_SOURCES src/model.c src/config.c src/net.c)
foreach(name server client observer)
    add_executable(kitchen_${name} src/${name}.c ${COMMON_SOURCES})
    target_include_directories(kitchen_${name} PRIVATE include)
    target_compile_definitions(kitchen_${name} PRIVATE _GNU_SOURCE)
    target_compile_options(kitchen_${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
enable_testing()
add_executable(model_test tests/model_test.c src/model.c)
target_include_directories(model_test PRIVATE include)
target_compile_definitions(model_test PRIVATE _GNU_SOURCE)
target_compile_options(model_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME model_invariants COMMAND model_test)
